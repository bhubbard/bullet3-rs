Quick Start Guide

Simulate a bouncing box dropped onto a static plane in 20 lines of Rust:

use bullet3::prelude::*;

fn main() {
    // 1. Create discrete dynamics world with standard gravity (9.81 m/s² down)
    let mut world = DiscreteDynamicsWorld::new()
        .with_gravity(Vector3::new(0.0, -9.81, 0.0));

    // 2. Add static ground plane at y = 0
    let ground_shape = Shape::StaticPlane(StaticPlaneShape::new(Vector3::Y, 0.0));
    let ground_info = RigidBodyConstructionInfo::new(0.0, ground_shape, Transform::IDENTITY);
    world.add_rigid_body(RigidBody::new(ground_info));

    // 3. Add dynamic falling box at height y = 10.0
    let box_shape = Shape::Box(BoxShape::new(Vector3::splat(0.5)));
    let box_tf = Transform::from_translation(Vector3::new(0.0, 10.0, 0.0));
    let box_info = RigidBodyConstructionInfo::new(1.0, box_shape, box_tf)
        .with_restitution(0.4)
        .with_friction(0.6);
    let box_id = world.add_rigid_body(RigidBody::new(box_info));

    // 4. Step simulation for 60 ticks (1.0s at 60 Hz)
    let dt = 1.0 / 60.0;
    for _ in 0..60 {
        world.step_simulation(dt, 10, dt);
    }

    let body = world.get_rigid_body(box_id).unwrap();
    println!("Settled Box Center Y: {:.3}m", body.transform.origin.y);
}

Built-in CLI Demos & Stress Tests

The `bullet3` crate compiles directly into a standalone benchmarking and verification CLI.

Box Stacking Stress Test stack

Verifies Gauss-Seidel constraint solver convergence and non-penetration stability for vertical rigid body stacks.

cargo run --release -- stack --boxes 5 --steps 180
Pendulum Joint Simulation pendulum

Verifies Point-to-Point spherical constraint rod distance conservation under gravity acceleration.

cargo run --release -- pendulum --steps 120
Elastic Billiard Collision billiards

Verifies 100% restitution momentum conservation and velocity transfer in zero-gravity space.

cargo run --release -- billiards --steps 60
Throughput Benchmark benchmark

Measures raw simulation throughput across multiple dynamic colliding bodies.

cargo run --release -- benchmark --steps 1000 --bodies 20
